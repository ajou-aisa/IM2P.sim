cmake_minimum_required(VERSION 3.20)
project(im2p_cycle_model VERSION 1.0 LANGUAGES C CXX)

option(IM2P_CYCLE_BUILD_TESTS "Build the separate cycle-model tests" ON)
option(IM2P_CYCLE_SANITIZE "Enable address/undefined behavior sanitizers" OFF)

# Deliberately independent of Rust, BSC, SBT, Verilator and the numerical backend.
set(cycle_sources
  ../common/gemmini_schedule.cpp
  scheduled_work.cpp
  control_engine.cpp
  execute_engine.cpp
  c_api.cpp)
add_library(im2p_cycle_objects OBJECT ${cycle_sources})
target_compile_features(im2p_cycle_objects PUBLIC cxx_std_20)
set_target_properties(im2p_cycle_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(im2p_cycle_objects PUBLIC ../include)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(im2p_cycle_objects PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_library(im2p_cycle_model STATIC $<TARGET_OBJECTS:im2p_cycle_objects>)
add_library(im2p_cycle_model_shared SHARED $<TARGET_OBJECTS:im2p_cycle_objects>)
set_target_properties(im2p_cycle_model_shared PROPERTIES OUTPUT_NAME im2p_cycle_model)
foreach(target IN ITEMS im2p_cycle_model im2p_cycle_model_shared)
  target_include_directories(${target} PUBLIC ../include)
  target_compile_features(${target} PUBLIC cxx_std_20)
endforeach()
if(IM2P_CYCLE_SANITIZE)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "Sanitizer option requires a Clang/GNU toolchain")
  endif()
  target_compile_options(im2p_cycle_objects PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  foreach(target IN ITEMS im2p_cycle_model im2p_cycle_model_shared)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    target_compile_options(${target} INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endforeach()
endif()

if(IM2P_CYCLE_BUILD_TESTS)
  enable_testing()
  add_executable(im2p_cycle_probe ../tests/cycle/probe.cpp)
  target_link_libraries(im2p_cycle_probe PRIVATE im2p_cycle_model)
  find_package(Threads REQUIRED)
  add_executable(im2p_cycle_unit ../tests/cycle/test_cycle_model.cpp)
  target_link_libraries(im2p_cycle_unit PRIVATE im2p_cycle_model Threads::Threads)
  add_executable(im2p_cycle_c_abi ../tests/cycle/test_c_api.c)
  set_target_properties(im2p_cycle_c_abi PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
  target_link_libraries(im2p_cycle_c_abi PRIVATE im2p_cycle_model)
  add_executable(im2p_cycle_runs ../tests/cycle/test_run_aware_cycle.cpp)
  target_link_libraries(im2p_cycle_runs PRIVATE im2p_cycle_model)
  add_executable(im2p_cycle_service ../tests/cycle/test_service_model.cpp)
  target_link_libraries(im2p_cycle_service PRIVATE im2p_cycle_model)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    foreach(target IN ITEMS im2p_cycle_probe im2p_cycle_unit im2p_cycle_c_abi im2p_cycle_runs im2p_cycle_service)
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endforeach()
  endif()
  add_test(NAME cycle_model_unit COMMAND im2p_cycle_unit)
  add_test(NAME cycle_model_c_abi COMMAND im2p_cycle_c_abi)
  add_test(NAME cycle_model_runs COMMAND im2p_cycle_runs)
  add_test(NAME cycle_model_service COMMAND im2p_cycle_service)
endif()
